using System;
using System.ComponentModel.DataAnnotations;
using System.ComponentModel.DataAnnotations.Schema;

namespace DentalClinicWPF.Models
{
    public class Invoice
    {
        [Key]
        public int InvoiceID { get; set; }

        [Required]
        public int PatientID { get; set; }
        [ForeignKey(nameof(PatientID))]
        public virtual Patient Patient { get; set; }

        // Puede venir de una Cita o de un Presupuesto (Opcionales)
        public int? AppointmentID { get; set; }
        public virtual Appointment Appointment { get; set; }

        public int? TreatmentPlanID { get; set; }
        public virtual TreatmentPlan TreatmentPlan { get; set; }

        public DateTime InvoiceDate { get; set; } = DateTime.Now;

        [Column(TypeName = "decimal(10, 2)")]
        public decimal TotalAmount { get; set; }

        [Column(TypeName = "decimal(10, 2)")]
        public decimal AmountPaid { get; set; }
        public decimal AmountReceived { get; set; }
        public decimal Change { get; set; }

        public string PaymentMethod { get; set; } = "Efectivo"; // Efectivo, Tarjeta, Transferencia

        [NotMapped]
        public bool IsPaid => AmountPaid >= TotalAmount;

        [NotMapped]
        public decimal BalanceDue => TotalAmount - AmountPaid;
    }
}

using System;
using System.Collections.ObjectModel;
using System.Linq;
using System.Windows;
using System.Windows.Input;
using CommunityToolkit.Mvvm.Input;
using DentalClinicWPF.Data;
using DentalClinicWPF.Models;
using DentalClinicWPF.ViewModels.Base;

namespace DentalClinicWPF.ViewModels.Appointment
{
    public class AddAppointmentViewModel : BaseViewModel
    {
        private readonly DentalClinicContext _context;

        private int _patientID;
        public int PatientID
        {
            get => _patientID;
            set
            {
                if (SetProperty(ref _patientID, value))
                {
                    ValidateProperty(nameof(PatientID));
                    NotifySaveCommand();
                }
            }
        }

        private int _dentistID;
        public int DentistID
        {
            get => _dentistID;
            set
            {
                if (SetProperty(ref _dentistID, value))
                {
                    ValidateProperty(nameof(DentistID));
                    NotifySaveCommand();
                }
            }
        }

        private int _treatmentID;
        public int TreatmentID
        {
            get => _treatmentID;
            set
            {
                if (SetProperty(ref _treatmentID, value))
                {
                    ValidateProperty(nameof(TreatmentID));
                    NotifySaveCommand();
                }
            }
        }

        private DateTime _appointmentDate = DateTime.Today;
        public DateTime AppointmentDate
        {
            get => _appointmentDate;
            set
            {
                if (SetProperty(ref _appointmentDate, value))
                {
                    ValidateProperty(nameof(AppointmentDate));
                    NotifySaveCommand();
                }
            }
        }

        private string _startTime = "09:00";
        public string StartTime
        {
            get => _startTime;
            set
            {
                if (SetProperty(ref _startTime, value))
                {
                    ValidateProperty(nameof(StartTime));
                    NotifySaveCommand();
                }
            }
        }

        private string _endTime = "10:00";
        public string EndTime
        {
            get => _endTime;
            set
            {
                if (SetProperty(ref _endTime, value))
                {
                    ValidateProperty(nameof(EndTime));
                    NotifySaveCommand();
                }
            }
        }

        private string _notes = string.Empty;
        public string Notes
        {
            get => _notes;
            set
            {
                if (SetProperty(ref _notes, value))
                {
                    ValidateProperty(nameof(Notes));
                    NotifySaveCommand();
                }
            }
        }

        public ObservableCollection<Models.Patient> Patients { get; }
        public ObservableCollection<Models.Dentist> Dentists { get; }
        public ObservableCollection<Models.Treatment> Treatments { get; }

        public ICommand SaveCommand { get; }
        public ICommand CancelCommand { get; }

        public AddAppointmentViewModel(DentalClinicContext context)
        {
            _context = context ?? throw new ArgumentNullException(nameof(context));

            Patients = new ObservableCollection<Models.Patient>(_context.Patient.ToList());
            Dentists = new ObservableCollection<Models.Dentist>(_context.Dentist.ToList());
            Treatments = new ObservableCollection<Models.Treatment>(_context.Treatment.ToList());

            SaveCommand = new RelayCommand(Save, CanSave);
            CancelCommand = new RelayCommand(Cancel);

            PropertyChanged += (s, e) => ValidateProperty(e.PropertyName);
        }

        protected override void ValidateProperty(string propertyName)
        {
            base.ValidateProperty(propertyName);

            switch (propertyName)
            {
                case nameof(PatientID):
                    if (PatientID == 0)
                        AddError(propertyName, "El paciente es obligatorio.");
                    break;

                case nameof(DentistID):
                    if (DentistID == 0)
                        AddError(propertyName, "El dentista es obligatorio.");
                    break;

                case nameof(TreatmentID):
                    if (TreatmentID == 0)
                        AddError(propertyName, "El tratamiento es obligatorio.");
                    break;

                case nameof(AppointmentDate):
                    if (AppointmentDate.Date < DateTime.Today)
                        AddError(propertyName, "La fecha de la cita no puede ser en el pasado.");
                    break;

                case nameof(StartTime):
                case nameof(EndTime):
                    ValidateTimeFormat(propertyName);
                    break;

                case nameof(Notes):
                    if (string.IsNullOrWhiteSpace(Notes))
                        AddError(propertyName, "Las notas son obligatorias.");
                    break;
            }
        }

        private void ValidateTimeFormat(string propertyName)
        {
            var time = propertyName == nameof(StartTime) ? StartTime : EndTime;

            if (!TimeSpan.TryParse(time, out var parsedTime))
                AddError(propertyName, "Formato de hora inválido. Use HH:mm.");
            else if (parsedTime < TimeSpan.FromHours(9) || parsedTime > TimeSpan.FromHours(18))
                AddError(propertyName, "La hora debe estar entre 09:00 y 18:00.");

            if (TimeSpan.TryParse(StartTime, out var start) && TimeSpan.TryParse(EndTime, out var end) && start >= end)
                AddError(nameof(StartTime), "La hora de inicio debe ser anterior a la hora de fin.");
        }

        private bool HasTimeConflict()
        {
            if (DentistID == 0) return false;
            if (!TimeSpan.TryParse(StartTime, out var start)) return false;
            if (!TimeSpan.TryParse(EndTime, out var end)) return false;

            var date = AppointmentDate.Date;

            return _context.Appointment
                .Where(a => a.DentistID == DentistID && a.AppointmentDate.Date == date)
                .AsEnumerable()
                .Any(a => a.StartTime < end && a.EndTime > start);
        }

        private bool CanSave()
        {
            return !HasErrors &&
                   PatientID > 0 &&
                   DentistID > 0 &&
                   TreatmentID > 0 &&
                   AppointmentDate >= DateTime.Today &&
                   TimeSpan.TryParse(StartTime, out var start) &&
                   TimeSpan.TryParse(EndTime, out var end) && start < end &&
                   !string.IsNullOrWhiteSpace(Notes);
        }

        private void Save()
        {
            // Check for time conflict first
            if (HasTimeConflict())
            {
                MessageBox.Show(
                    "El dentista ya tiene una cita en ese horario.\n\n" +
                    "Por favor elige otra hora o fecha.",
                    "Horario ocupado",
                    MessageBoxButton.OK,
                    MessageBoxImage.Warning);
                return;
            }

            if (!CanSave())
            {
                MessageBox.Show("Corrige los errores de validación antes de guardar.", "Error de Validación", MessageBoxButton.OK, MessageBoxImage.Warning);
                return;
            }

            try
            {
                var appointment = new Models.Appointment
                {
                    PatientID = PatientID,
                    DentistID = DentistID,
                    TreatmentID = TreatmentID,
                    AppointmentDate = AppointmentDate,
                    StartTime = TimeSpan.Parse(StartTime),
                    EndTime = TimeSpan.Parse(EndTime),
                    Notes = Notes
                };

                _context.Appointment.Add(appointment);
                _context.SaveChanges();

                MessageBox.Show("¡Cita agregada correctamente!", "Éxito", MessageBoxButton.OK, MessageBoxImage.Information);
                CloseWindow(true);
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Error al guardar la cita: {ex.Message}", "Error", MessageBoxButton.OK, MessageBoxImage.Error);
            }
        }

        private void Cancel() => CloseWindow(false);

        private void NotifySaveCommand() => ((RelayCommand)SaveCommand).NotifyCanExecuteChanged();

        private void CloseWindow(bool dialogResult)
        {
            foreach (Window window in Application.Current.Windows)
            {
                if (window.DataContext == this)
                {
                    window.DialogResult = dialogResult;
                    window.Close();
                    break;
                }
            }
        }
    }
}

        public ObservableCollection<Models.Room> Rooms { get; set; }

        public ICommand CancelCommand { get; }

        public event EventHandler<DataErrorsChangedEventArgs>? ErrorsChanged;
        public bool HasErrors => _errors.Any();

        public AddEquipmentViewModel(DentalClinicContext context)
        {
            _context = context ?? throw new ArgumentNullException(nameof(context));

            Equipment = new Models.Equipment
            {
                Name = string.Empty,
                Model = string.Empty,
                SerialNumber = string.Empty,
                PurchaseDate = null,
                LastServiceDate = null
            };

            Rooms = new ObservableCollection<Models.Room>(_context.Room.ToList());
            SaveCommand = new RelayCommand(Save, CanSave);
            CancelCommand = new RelayCommand(Cancel);
        }

        public IEnumerable GetErrors(string? propertyName)
        {
            if (string.IsNullOrEmpty(propertyName))
                return _errors.SelectMany(err => err.Value);
            else if (_errors.ContainsKey(propertyName))
                return _errors[propertyName];

            return Enumerable.Empty<string>();
        }

        private void ValidateProperty(string propertyName)
        {
            if (string.IsNullOrEmpty(propertyName)) return;

            _errors.Remove(propertyName);

            switch (propertyName)
            {
                case nameof(Equipment.Name):
                    if (string.IsNullOrWhiteSpace(Equipment.Name))
                        AddError(propertyName, "El nombre es obligatorio.");
                    break;
                case nameof(Equipment.Model):
                    if (string.IsNullOrWhiteSpace(Equipment.Model))
                        AddError(propertyName, "El modelo es obligatorio.");
                    break;
                case nameof(Equipment.SerialNumber):
                    if (!string.IsNullOrWhiteSpace(Equipment.SerialNumber) &&
                        !Regex.IsMatch(Equipment.SerialNumber, @"^[a-zA-Z0-9-]+$"))
                        AddError(propertyName, "El número de serie solo puede contener caracteres alfanuméricos o guiones.");
                    break;
                case nameof(Equipment.PurchaseDate):
                    if (Equipment.PurchaseDate > DateTime.Now)
                        AddError(propertyName, "La fecha de compra no puede ser futura.");
                    break;
                case nameof(Equipment.LastServiceDate):
                    if (Equipment.LastServiceDate > DateTime.Now)
                        AddError(propertyName, "La fecha del último servicio no puede ser futura.");
                    break;
            }

            ErrorsChanged?.Invoke(this, new DataErrorsChangedEventArgs(propertyName));
            ((RelayCommand)SaveCommand).NotifyCanExecuteChanged();
        }

        private void AddError(string propertyName, string error)
        {
            if (!_errors.ContainsKey(propertyName))
                _errors[propertyName] = new List<string>();

            _errors[propertyName].Add(error);
        }

        private bool CanSave() => !HasErrors;

        private void Save()
        {
            if (HasErrors)
            {
                MessageBox.Show("Corrige los errores de validación antes de guardar.", "Error de Validación", MessageBoxButton.OK, MessageBoxImage.Warning);
                return;
            }

            try
            {
                Equipment.RoomID = SelectedRoom?.RoomID;

                _context.Equipment.Add(Equipment);
                _context.SaveChanges();

                MessageBox.Show("¡Equipo agregado correctamente!", "Éxito", MessageBoxButton.OK, MessageBoxImage.Information);
                CloseWindow();
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Error al guardar el equipo: {ex.Message}", "Error", MessageBoxButton.OK, MessageBoxImage.Error);
            }
        }

using System;
using System.Collections.Generic;

{
    public class AddScheduleViewModel : BaseViewModel
    {
        private readonly DentalClinicContext _context;

        private Models.Schedule _schedule;
        public Models.Schedule Schedule
        {
            get => _schedule;
            set
            {
                _schedule = value;
                OnPropertyChanged();
            }
        }

        public List<Models.Dentist> Dentists { get; }

        public ICommand SaveCommand { get; }
        public ICommand CancelCommand { get; }

        public AddScheduleViewModel(DentalClinicContext context)
        {
            _context = context ?? throw new ArgumentNullException(nameof(context));

            Schedule = new Models.Schedule
            {
                StartDate = DateTime.Today,
                EndDate = DateTime.Today.AddDays(4), // Default to a 5-day week
                IsDayOff = false
                // StartTime and EndTime are set to default values in the model
            };

            // Retrieve the list of dentists
            Dentists = _context.Dentist.ToList();

            SaveCommand = new RelayCommand(Save);
            CancelCommand = new RelayCommand(Cancel);
        }

        private void Save()
        {
            // Validate that work dates are Monday to Friday
            if (Schedule.StartDate.DayOfWeek == DayOfWeek.Saturday || Schedule.StartDate.DayOfWeek == DayOfWeek.Sunday ||
                Schedule.EndDate.DayOfWeek == DayOfWeek.Saturday || Schedule.EndDate.DayOfWeek == DayOfWeek.Sunday)
            {
                MessageBox.Show("Los empleados no trabajan los fines de semana.", "Error de Validación", MessageBoxButton.OK, MessageBoxImage.Warning);
                return;
            }

            // Ensure the date range is logical
            if (Schedule.StartDate > Schedule.EndDate)
            {
                MessageBox.Show("La fecha de inicio debe ser anterior a la fecha de fin.", "Error de Validación", MessageBoxButton.OK, MessageBoxImage.Warning);
                return;
            }

            // Check if a schedule for this dentist and date range already exists
            var overlappingSchedule = _context.Schedule
                .Where(s => s.DentistID == Schedule.DentistID)
                .Any(s => s.StartDate <= Schedule.EndDate && s.EndDate >= Schedule.StartDate);

            if (overlappingSchedule)
            {
                MessageBox.Show("Ya existe un horario para este dentista en ese rango de fechas.", "Error de Validación", MessageBoxButton.OK, MessageBoxImage.Warning);
                return;
            }

            // Validate the selected dentist
            var dentist = Dentists.FirstOrDefault(d => d.DentistID == Schedule.DentistID);
            if (dentist == null)
            {
                MessageBox.Show("El dentista seleccionado no existe.", "Error de Validación", MessageBoxButton.OK, MessageBoxImage.Warning);
                return;
            }

            _context.Schedule.Add(Schedule);
            _context.SaveChanges();

            // Set DialogResult to true to indicate success
            SetDialogResult(true);
        }

        private void Cancel()
        {
            // Set DialogResult to false to indicate cancellation
            SetDialogResult(false);
        }

        private void SetDialogResult(bool result)
        {
            foreach (Window window in Application.Current.Windows)
            {
                if (window.DataContext == this)
                {
                    window.DialogResult = result;
                    window.Close();
                    break;
                }
            }
        }

                MessageBox.Show($"Error al guardar el tratamiento: {ex.Message}", "Error", MessageBoxButton.OK, MessageBoxImage.Error);
            }
        }

        private void Cancel() => CloseWindow(false);

        private void NotifySaveCommand()
        {
            if (SaveCommand is RelayCommand relayCommand)
                relayCommand.NotifyCanExecuteChanged();
        }

        private void CloseWindow(bool dialogResult)
        {
            foreach (Window window in Application.Current.Windows)
            {
                if (window.DataContext == this)
                {
                    window.DialogResult = dialogResult;
                    window.Close();
                    break;
                }
            }
        }
    }

using System;
using System.Collections.ObjectModel;
using System.Linq;
using System.Windows;
using CommunityToolkit.Mvvm.Input;
using DentalClinicWPF.Data;
using DentalClinicWPF.ViewModels.Base;

namespace DentalClinicWPF.ViewModels.TreatmentPlan
{
    public class AddTreatmentPlanViewModel : BaseViewModel
    {
        private readonly DentalClinicContext _context;
        private readonly Models.Patient _patient;

        public ObservableCollection<Models.Treatment> AvailableTreatments { get; set; }
        public ObservableCollection<Models.Dentist> AvailableDentists { get; set; }
        public ObservableCollection<TreatmentPlanItemViewModel> Items { get; set; }

        private Models.Treatment _selectedTreatment;
        public Models.Treatment SelectedTreatment
        {
            get => _selectedTreatment;
            set { _selectedTreatment = value; OnPropertyChanged(nameof(SelectedTreatment)); }
        }

        private Models.Dentist _selectedDentist;
        public Models.Dentist SelectedDentist
        {
            get => _selectedDentist;
            set { _selectedDentist = value; OnPropertyChanged(nameof(SelectedDentist)); }
        }

        private int _quantity = 1;
        public int Quantity
        {
            get => _quantity;
            set { _quantity = value; OnPropertyChanged(nameof(Quantity)); }
        }

        private string _observations;
        public string Observations
        {
            get => _observations;
            set { _observations = value; OnPropertyChanged(nameof(Observations)); }
        }

        public decimal TotalAmount => Items.Sum(x => x.Subtotal);

        public IRelayCommand AddItemCommand { get; }
        public IRelayCommand<TreatmentPlanItemViewModel> RemoveItemCommand { get; }
        public IRelayCommand SaveCommand { get; }

        public AddTreatmentPlanViewModel(DentalClinicContext context, Models.Patient patient)
        {
            _context = context ?? throw new ArgumentNullException(nameof(context));
            _patient = patient ?? throw new ArgumentNullException(nameof(patient));

            Items = new ObservableCollection<TreatmentPlanItemViewModel>();
            AvailableTreatments = new ObservableCollection<Models.Treatment>(_context.Treatment.ToList());
            AvailableDentists = new ObservableCollection<Models.Dentist>(_context.Dentist.ToList());
            SelectedDentist = AvailableDentists.FirstOrDefault();

            AddItemCommand = new RelayCommand(AddItem);
            RemoveItemCommand = new RelayCommand<TreatmentPlanItemViewModel>(RemoveItem);
            SaveCommand = new RelayCommand(Save);
        }

        private void AddItem()
        {
            if (SelectedTreatment == null || Quantity <= 0)
            {
                MessageBox.Show("Seleccione un tratamiento y una cantidad válida.", "Aviso", MessageBoxButton.OK, MessageBoxImage.Warning);
                return;
            }

            var existing = Items.FirstOrDefault(x => x.Treatment.TreatmentID == SelectedTreatment.TreatmentID);
            if (existing != null)
            {
                existing.Quantity += Quantity;
                // Forzar actualización en UI
                var index = Items.IndexOf(existing);
                Items[index] = existing;
            }
            else
            {
                Items.Add(new TreatmentPlanItemViewModel(SelectedTreatment, Quantity));
            }

            Quantity = 1;
            OnPropertyChanged(nameof(TotalAmount));
        }

        private void RemoveItem(TreatmentPlanItemViewModel item)
        {
            if (item != null && Items.Contains(item))
            {
                Items.Remove(item);
                OnPropertyChanged(nameof(TotalAmount));
            }
        }

        private void Save()
        {
            if (!Items.Any())
            {
                MessageBox.Show("Debe agregar al menos un tratamiento al presupuesto.", "Aviso", MessageBoxButton.OK, MessageBoxImage.Warning);
                return;
            }
            if (SelectedDentist == null)
            {
                MessageBox.Show("Debe seleccionar un odontólogo.", "Aviso", MessageBoxButton.OK, MessageBoxImage.Warning);
                return;
            }

            try
            {
                var plan = new Models.TreatmentPlan
                {
                    PatientID = _patient.PatientID,
                    DentistID = SelectedDentist.DentistID,
                    CreationDate = DateTime.Now,
                    Status = "Borrador",
                    TotalAmount = TotalAmount,
                    Observations = Observations,
                    Details = string.Empty // <- Requerido por el esquema antiguo de SQLite
                };

                // Guardar el plan primero para generar el TreatmentPlanID
                _context.TreatmentPlan.Add(plan);
                _context.SaveChanges();

                // Ahora guardar los items con el ID del plan ya generado
                foreach (var item in Items)
                {
                    var entity = item.ToEntity();
                    entity.TreatmentPlanID = plan.TreatmentPlanID;
                    _context.TreatmentPlanItem.Add(entity);
                }
                _context.SaveChanges();
                
                MessageBox.Show("Presupuesto guardado correctamente.", "Éxito", MessageBoxButton.OK, MessageBoxImage.Information);
                // Cerrar ventana (requiere que el code-behind escuche este evento o se use un DialogResult)
                if (Application.Current.Windows.OfType<Window>().SingleOrDefault(x => x.IsActive) is Window w) { w.DialogResult = true; }
            }
            catch (Exception ex)
            {
                var innerMsg = ex.InnerException != null ? ex.InnerException.Message : "";
                MessageBox.Show($"Error al guardar:\n{ex.Message}\n\nDetalle:\n{innerMsg}", "Error", MessageBoxButton.OK, MessageBoxImage.Error);
            }
        }
    }
}
